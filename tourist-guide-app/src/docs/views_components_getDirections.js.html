/**
 * Fetches directions from the Google Directions API and processes the route data.
 *
 * This function sends a request to the Google Directions API to get driving directions
 * from the current location to the specified destination. It processes the response
 * to extract the coordinates of the route steps and updates the state with the first
 * route's steps.
 *
 * If either the location or destination is missing, an alert is shown to the user.
 *
 * @async
 * @function
 * @throws {Error} If the API request fails or if there is a missing location or destination.
 * @returns {void} This function doesn't return a value; it updates the route state with the directions data.
 *
 * @example
 * const location = { latitude: 37.7749, longitude: -122.4194 }; // San Francisco
 * const destination = 'Los Angeles';
 * getDirections(location, destination);
 */
const getDirections = async () => {
  if (!location || !destination) {
    Alert.alert('Missing information', 'Please make sure both location and destination are provided.');
    return;
  }

  // Prepare the API request URL for Google Directions API
  const directionsUrl = `https://maps.googleapis.com/maps/api/directions/json?origin=${location.latitude},${location.longitude}&destination=${destination}&mode=driving&alternatives=true&key=YOUR_GOOGLE_MAPS_API_KEY`;

  try {
    const response = await axios.get(directionsUrl);
    const routes = response.data.routes;

    // Loop through each route and extract coordinates
    const allRoutes = routes.map(route => {
      return route.legs[0].steps.map(step => ({
        latitude: step.end_location.lat,
        longitude: step.end_location.lng,
      }));
    });

    setRoute(allRoutes[0]); // Default to first route

    // Allow user to choose an alternative route if available
    // For example, you could store these routes in a state and display options in the UI
  } catch (error) {
    Alert.alert('Error', 'Unable to fetch directions.');
  }
};
